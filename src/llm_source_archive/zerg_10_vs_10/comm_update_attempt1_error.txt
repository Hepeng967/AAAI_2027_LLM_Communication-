def communication(o):
    """
    Constructs a message for each agent to be broadcast to teammates.

    Args:
        o: observation tensor, shape (2, 10, 188)

    Returns:
        messages: tensor of shape (2, 10, 34) containing the outgoing message
                  for each agent. The message will be appended to recipients'
                  observations by the framework.
    """
    batch_size, n_agents, _ = o.shape

    # --- 1. Self information (type, health, absolute position) ---
    # own_health: 156, unit_type bits: 157-159, own_pos_x: 160, own_pos_y: 161
    self_info = o[..., 156:162]   # shape (2, 10, 6)

    # --- 2. Last action one-hot (16 dims: dims 162-177) ---
    last_action = o[..., 162:178]  # shape (2, 10, 16)

    # --- 3. Enemy information ---
    # All enemy blocks are in dims 4..83 (80 values) → (2,10,80) → reshape to (2,10,10,8)
    enemy_block = o[..., 4:84].reshape(batch_size, n_agents, 10, 8)

    # Extract per-enemy fields
    rel_x = enemy_block[..., 2]          # (2,10,10)
    rel_y = enemy_block[..., 3]          # (2,10,10)
    health = enemy_block[..., 4]         # (2,10,10)
    type_bits = enemy_block[..., 5:8]    # (2,10,10,3)

    # Compute enemy absolute positions using own absolute position
    own_pos_x = o[..., 160:161]  # (2,10,1)
    own_pos_y = o[..., 161:162]  # (2,10,1)
    abs_x = own_pos_x + rel_x    # (2,10,10) broadcast
    abs_y = own_pos_y + rel_y    # (2,10,10)

    # Select two enemies with smallest health (priority targets)
    valid_mask = health > 0                     # alive and observed
    health_for_sort = health.clone()
    health_for_sort[~valid_mask] = float('inf')
    topk_health, topk_indices = th.topk(health_for_sort, k=2, dim=2, largest=False)  # (2,10,2)

    # Gather enemy features at the selected indices
    # type_bits: (2,10,10,3) → gather along dim=2
    indices_3d = topk_indices.unsqueeze(-1).expand(-1, -1, -1, 3)  # (2,10,2,3)
    sel_type_bits = th.gather(type_bits, dim=2, index=indices_3d)  # (2,10,2,3)

    # health, abs_x, abs_y: (2,10,10) → gather along dim=2
    indices_2d = topk_indices                                       # (2,10,2)
    sel_health = th.gather(health, dim=2, index=indices_2d)        # (2,10,2)
    sel_abs_x  = th.gather(abs_x, dim=2, index=indices_2d)
    sel_abs_y  = th.gather(abs_y, dim=2, index=indices_2d)

    # Stack into one vector: [type_bits (3), health (1), abs_x (1), abs_y (1)] per enemy
    sel_health = sel_health.unsqueeze(-1)   # (2,10,2,1)
    sel_abs_x  = sel_abs_x.unsqueeze(-1)
    sel_abs_y  = sel_abs_y.unsqueeze(-1)
    enemy_feat = th.cat([sel_type_bits, sel_health, sel_abs_x, sel_abs_y], dim=-1)  # (2,10,2,6)

    # Zero out invalid entries (where original health was inf)
    valid_selected = (topk_health < float('inf')).unsqueeze(-1).float()  # (2,10,2,1)
    enemy_feat = enemy_feat * valid_selected                            # broadcast over last dim

    # Flatten to (2,10,12)
    enemy_info = enemy_feat.reshape(batch_size, n_agents, 12)

    # --- 4. Compose final message ---
    message = th.cat([self_info, last_action, enemy_info], dim=-1)  # (2,10,34)

    return message
